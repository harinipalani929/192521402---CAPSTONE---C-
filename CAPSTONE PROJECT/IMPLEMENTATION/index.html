#include <algorithm>
#include <cmath>
#include <numeric>
#include <string>
#include <vector>

// --- MODULE 1: Data Ingestion & Preprocessing ---
struct PreprocessedSeries {
    std::vector<double> cleanClose;
    std::vector<double> ema12;
    std::vector<double> returns;
};

PreprocessedSeries preprocessTicks(const std::vector<double>& raw, double zThreshold = 2.5) {
    PreprocessedSeries out;
    if (raw.empty()) return out;

    double mean = std::accumulate(raw.begin(), raw.end(), 0.0) / raw.size();
    double sqSum = 0.0;
    for (double v : raw) sqSum += (v - mean) * (v - mean);
    double stdev = std::sqrt(sqSum / raw.size());

    out.cleanClose.reserve(raw.size());
    for (double v : raw) {
        double clipped = std::clamp(v, mean - zThreshold * stdev, mean + zThreshold * stdev);
        out.cleanClose.push_back(clipped);
    }

    double k = 2.0 / (12.0 + 1.0);
    out.ema12.push_back(out.cleanClose[0]);
    for (size_t i = 1; i < out.cleanClose.size(); ++i) {
        out.ema12.push_back(out.cleanClose[i] * k + out.ema12.back() * (1.0 - k));
        out.returns.push_back((out.cleanClose[i] - out.cleanClose[i - 1]) / out.cleanClose[i - 1]);
    }
    return out;
}

// --- MODULE 2: Predictive Modelling & Analytical Engine ---
struct ForecastPoint {
    int day;
    double predicted;
    double lower;
    double upper;
};

std::vector<ForecastPoint> forecastSeries(const PreprocessedSeries& data, int horizon, double zConf) {
    std::vector<ForecastPoint> result;
    double last = data.cleanClose.back();
    double drift = std::accumulate(data.returns.begin(), data.returns.end(), 0.0) / data.returns.size();
    double vol = 0.015;

    for (int h = 1; h <= horizon; ++h) {
        double pred = last * std::pow(1.0 + drift * 0.65, h);
        double band = pred * vol * std::sqrt(static_cast<double>(h)) * zConf;
        result.push_back({h, pred, pred - band, pred + band});
    }
    return result;
}
